add_executable(thermique thermique.cxx)
target_link_libraries(thermique PRIVATE mfem-mgis::MFEMMGIS)
add_mfront_library(thermique-behaviours
  "${MFEM_MGIS_SOURCE_DIR}/tests/StationaryNonLinearHeatTransfer.mfront")
configure_file("${MFEM_MGIS_SOURCE_DIR}/tests/beam-tet.mesh"
  "${CMAKE_CURRENT_BINARY_DIR}/beam-tet.mesh" COPYONLY)

# short tests: two steps on a coarse mesh; the linear solvers give the same
# temperature
set(norm 7227.044268)
add_test(NAME thermique-umfpack
  COMMAND ./thermique -r 1 -n 2 -ls umfpack -no-p -en ${norm})
add_test(NAME thermique-mumps
  COMMAND mpirun -n 2 ./thermique -r 1 -n 2 -ls mumps -en ${norm})
add_test(NAME thermique-gmres-amg
  COMMAND mpirun -n 2 ./thermique -r 1 -n 2 -ls gmres-amg -en ${norm})
# the comparison detects a wrong solution
add_test(NAME thermique-wrong-norm
  COMMAND ./thermique -r 1 -n 2 -ls umfpack -no-p -en 1)
set_tests_properties(thermique-wrong-norm PROPERTIES WILL_FAIL TRUE)

# the integration by batches with the gradients, the inner forces and the
# stiffness matrices computed by kernels (options given by the environment)
set(kernels
  "MFEM_MGIS_BEHAVIOUR_INTEGRATION=Batch;MFEM_MGIS_GRADIENTS_COMPUTATION=Kernels;MFEM_MGIS_INNER_FORCES_COMPUTATION=Kernels")
add_test(NAME thermique-noyaux
  COMMAND mpirun -n 2 ./thermique -r 1 -n 2 -ls gmres-amg -en ${norm})
set_tests_properties(thermique-noyaux PROPERTIES ENVIRONMENT
  "${kernels};MFEM_MGIS_TANGENT_OPERATOR_COMPUTATION=Kernels")
# convective exchange at the second end (Robin boundary condition, boundary
# face integrator of the non linear form): same temperature point by point
# and by kernels, the contributions of the boundary face integrator being
# added to the inner forces and to the local matrix computed by kernels
set(robin_norm 5502.767622)
add_test(NAME thermique-robin-umfpack
  COMMAND ./thermique -r 1 -n 2 -ls umfpack -no-p -hc 1 -en ${robin_norm})
add_test(NAME thermique-robin-noyaux
  COMMAND mpirun -n 2 ./thermique -r 1 -n 2 -ls gmres-amg -hc 1
          -en ${robin_norm})
set_tests_properties(thermique-robin-noyaux PROPERTIES ENVIRONMENT
  "${kernels};MFEM_MGIS_TANGENT_OPERATOR_COMPUTATION=Kernels")
# the tangent operator applied without assembly, with the GMRES of MFEM and a
# Jacobi preconditioner
add_test(NAME thermique-sans-matrice
  COMMAND mpirun -n 2 ./thermique -r 1 -n 2 -ls gmres-jacobi -lt 1e-12
          -en ${norm})
set_tests_properties(thermique-sans-matrice PROPERTIES ENVIRONMENT
  "${kernels};MFEM_MGIS_TANGENT_OPERATOR_COMPUTATION=MatrixFree")

# the behaviour generated by the generic-parallel interface on the host
# (openmp and stlpar backends) and on the GPU (cuda backend), all the
# computations being made by kernels
add_generic_parallel_mfront_libraries(thermique-behaviours CUDA
  "${MFEM_MGIS_SOURCE_DIR}/tests/StationaryNonLinearHeatTransfer.mfront")
foreach(_backend ${thermique-behaviours_BACKENDS})
  set(_library "${CMAKE_CURRENT_BINARY_DIR}/generic-parallel-${_backend}/src/libGenericParallelBehaviour-${_backend}.so")
  if(_backend STREQUAL "cuda")
    add_test(NAME thermique-gpu
      COMMAND mpirun -n 1 ./thermique -r 1 -n 2 -ls gmres-amg -bi Batch
              --device cuda -l "${_library}" -en ${norm})
    set_tests_properties(thermique-gpu PROPERTIES ENVIRONMENT
      "${kernels};MFEM_MGIS_TANGENT_OPERATOR_COMPUTATION=Kernels"
      RESOURCE_LOCK gpu)
  else()
    add_test(NAME thermique-batch-${_backend}
      COMMAND mpirun -n 2 ./thermique -r 1 -n 2 -ls gmres-amg -bi Batch
              -l "${_library}" -en ${norm})
  endif()
endforeach()
