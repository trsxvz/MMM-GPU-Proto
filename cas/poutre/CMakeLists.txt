add_executable(poutre poutre.cxx)
target_link_libraries(poutre PRIVATE mfem-mgis::MFEMMGIS)
add_mfront_library(poutre-behaviours
  "${MFEM_MGIS_SOURCE_DIR}/tests/IsotropicLinearHardeningPlasticity.mfront")
configure_file("${MFEM_MGIS_SOURCE_DIR}/tests/beam-tet.mesh"
  "${CMAKE_CURRENT_BINARY_DIR}/beam-tet.mesh" COPYONLY)

# short tests: two steps on a coarse mesh; the three linear solvers give the
# same solution
set(norm 3.264041277)
add_test(NAME poutre-pcg-amg
  COMMAND mpirun -n 2 ./poutre -r 1 -n 2 -ls pcg-amg -en ${norm})
add_test(NAME poutre-mumps
  COMMAND mpirun -n 2 ./poutre -r 1 -n 2 -ls mumps -en ${norm})
add_test(NAME poutre-umfpack
  COMMAND ./poutre -r 1 -n 2 -ls umfpack -no-p -en ${norm})
# the comparison detects a wrong solution
add_test(NAME poutre-wrong-norm
  COMMAND ./poutre -r 1 -n 2 -ls umfpack -no-p -en 3.3)
set_tests_properties(poutre-wrong-norm PROPERTIES WILL_FAIL TRUE)

# the same behaviour generated by the generic-parallel interface on the host
# (openmp and stlpar backends), if TFEL provides the configuration files
get_filename_component(_mfront_bin "${MFRONT}" DIRECTORY)
foreach(backend_configuration openmp:config-openmp-host-gcc.json
                              stlpar:config-stlpar-parunseq-gcc.json)
  string(REPLACE ":" ";" _bc "${backend_configuration}")
  list(GET _bc 0 _backend)
  list(GET _bc 1 _configuration)
  if(EXISTS
     "${_mfront_bin}/../share/tfel/mfront/configuration-files/${_configuration}")
    set(_directory "${CMAKE_CURRENT_BINARY_DIR}/generic-parallel-${_backend}")
    file(MAKE_DIRECTORY "${_directory}")
    add_custom_command(
      OUTPUT "${_directory}/src/libGenericParallelBehaviour-${_backend}.so"
      COMMAND ${MFRONT} --obuild --interface=generic-parallel
              "--configuration-file=${_configuration}"
              "${MFEM_MGIS_SOURCE_DIR}/tests/IsotropicLinearHardeningPlasticity.mfront"
      WORKING_DIRECTORY "${_directory}"
      DEPENDS "${MFEM_MGIS_SOURCE_DIR}/tests/IsotropicLinearHardeningPlasticity.mfront"
      COMMENT "MFront: behaviour of the beam, generic-parallel (${_backend})")
    add_custom_target(poutre-behaviours-${_backend} ALL
      DEPENDS "${_directory}/src/libGenericParallelBehaviour-${_backend}.so")
    add_test(NAME poutre-batch-${_backend}
      COMMAND mpirun -n 2 ./poutre -r 1 -n 2 -ls pcg-amg -bi Batch
              -l "${_directory}/src/libGenericParallelBehaviour-${_backend}.so"
              -en ${norm})
  endif()
endforeach()
