# short tests: two steps on a coarse mesh; the three linear solvers give the
# same solution
set(norm 3.264041277)
add_test(NAME poutre-pcg-amg
  COMMAND mpirun -n 2 ./poutre -r 1 -n 2 -ls pcg-amg -en ${norm})
add_test(NAME poutre-mumps
  COMMAND mpirun -n 2 ./poutre -r 1 -n 2 -ls mumps -en ${norm})
add_test(NAME poutre-umfpack
  COMMAND ./poutre -r 1 -n 2 -ls umfpack -no-p -en ${norm})
# the comparison detects a wrong solution
add_test(NAME poutre-wrong-norm
  COMMAND ./poutre -r 1 -n 2 -ls umfpack -no-p -en 3.3)
set_tests_properties(poutre-wrong-norm PROPERTIES WILL_FAIL TRUE)

# the integration by batches with the gradients, the inner forces and the
# stiffness matrices computed by kernels (options given by the environment)
add_test(NAME poutre-noyaux
  COMMAND mpirun -n 2 ./poutre -r 1 -n 2 -ls pcg-amg -en ${norm})

  "MFEM_MGIS_BEHAVIOUR_INTEGRATION=Batch;MFEM_MGIS_GRADIENTS_COMPUTATION=Kernels;MFEM_MGIS_INNER_FORCES_COMPUTATION=Kernels;MFEM_MGIS_TANGENT_OPERATOR_COMPUTATION=Kernels")

  "MFEM_MGIS_BEHAVIOUR_INTEGRATION=Batch;MFEM_MGIS_GRADIENTS_COMPUTATION=Kernels;MFEM_MGIS_INNER_FORCES_COMPUTATION=Kernels;MFEM_MGIS_TANGENT_OPERATOR_COMPUTATION=MatrixFree")

# the same behaviour generated by the generic-parallel interface on the host
# (openmp and stlpar backends), if TFEL provides the configuration files, and
# on the GPU (cuda backend), integrated on the copies of the arrays on the
# GPU, all the computations being made by kernels
add_generic_parallel_mfront_libraries(poutre-behaviours CUDA
  "${MFEM_MGIS_SOURCE_DIR}/tests/IsotropicLinearHardeningPlasticity.mfront")
# the other backends for the GPU which work in mfem-mgis (see
# cas/mesures-backends-gpu.sh): cuda by clang, hip by clang and by hipcc on
# the NVIDIA platform, kokkos (Kokkos being initialized by the option
# --kokkos); short tests on the GPU
foreach(_gpu_backend cuda-clang:cuda hip-clang-nvidia-platform:hip
                     hip-hipcc:hip kokkos:kokkos)
  string(REPLACE ":" ";" _gb "${_gpu_backend}")
  list(GET _gb 0 _name)
  list(GET _gb 1 _suffix)
  if((_suffix STREQUAL "kokkos") AND (NOT TARGET proto-kokkos-runtime))
    continue()
  endif()
  add_generic_parallel_mfront_library(poutre-behaviours ${_name}
    config-${_name}.json ${_suffix}
    "${MFEM_MGIS_SOURCE_DIR}/tests/IsotropicLinearHardeningPlasticity.mfront")
  if(NOT poutre-behaviours_${_name}_FOUND)
    continue()
  endif()
  set(_options)
  if(_suffix STREQUAL "kokkos")
    set(_options --kokkos)
  endif()
  add_test(NAME poutre-gpu-${_name}
    COMMAND mpirun -n 2 ./poutre -r 1 -n 2 -ls pcg-amg -bi Batch
            --device cuda ${_options}
            -l "${CMAKE_CURRENT_BINARY_DIR}/generic-parallel-${_name}/src/libGenericParallelBehaviour-${_suffix}.so"
            -en ${norm})
  set_tests_properties(poutre-gpu-${_name} PROPERTIES ENVIRONMENT
    "MFEM_MGIS_GRADIENTS_COMPUTATION=Kernels;MFEM_MGIS_INNER_FORCES_COMPUTATION=Kernels;MFEM_MGIS_TANGENT_OPERATOR_COMPUTATION=Kernels"
    RESOURCE_LOCK gpu)
endforeach()
foreach(_backend ${poutre-behaviours_BACKENDS})
  set(_library "${CMAKE_CURRENT_BINARY_DIR}/generic-parallel-${_backend}/src/libGenericParallelBehaviour-${_backend}.so")
  if(_backend STREQUAL "cuda")
    add_test(NAME poutre-gpu
      COMMAND mpirun -n 2 ./poutre -r 1 -n 2 -ls pcg-amg -bi Batch
              --device cuda -l "${_library}" -en ${norm})
    # each process on the GPU reserves a memory pool of 4 GiB (Umpire): the
    # tests on the GPU are not run at the same time
    set_tests_properties(poutre-gpu PROPERTIES ENVIRONMENT
      "MFEM_MGIS_GRADIENTS_COMPUTATION=Kernels;MFEM_MGIS_INNER_FORCES_COMPUTATION=Kernels;MFEM_MGIS_TANGENT_OPERATOR_COMPUTATION=Kernels"
      RESOURCE_LOCK gpu)
  else()
    add_test(NAME poutre-batch-${_backend}
      COMMAND mpirun -n 2 ./poutre -r 1 -n 2 -ls pcg-amg -bi Batch
              -l "${_library}" -en ${norm})
  endif()
endforeach()
