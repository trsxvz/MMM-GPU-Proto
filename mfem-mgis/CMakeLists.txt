cmake_minimum_required(VERSION 3.8)
project(mfem-mgis
  VERSION 1.1.0
  DESCRIPTION "The aim of the MFEM/MGIS project is to provide a C++ library to build advanced mechanical simulation based on MFront behaviours with MFEM.")

include(cmake/modules/mfem-mgis.cmake)
list(APPEND CMAKE_MODULE_PATH ${PROJECT_SOURCE_DIR}/cmake/modules)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ===========================
# === CMake customization ===
# ===========================
set_property(CACHE CMAKE_BUILD_TYPE PROPERTY STRINGS Release Coverage RelWithDebInfo Debug)
if (NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE "Release" CACHE STRING "" FORCE)
endif()

# (must be placed *before* any add_subdirectory, cmake bug ?)
include(CTest)
enable_testing()

# portable-build
option(enable-portable-build "Produce binary that can be shared between various machine (same architecture, same gcc version, different processors" OFF)
option(BENCHMARKS "Compile benchmarks" OFF)
option(enable-gpu-tests "Add the tests which require a GPU (when MFEM is compiled with CUDA)" ON)
option(enable-amgx "Solvers of the AmgX library of NVIDIA (requires MFEM compiled with CUDA, AmgX given by AMGX_DIR)" OFF)
option(enable-kokkos "Initialization of Kokkos by mfem_mgis::initialize on demand (option --kokkos or variable MFEM_MGIS_KOKKOS), for the behaviours generated by the kokkos backend of the generic-parallel interface of MFront" OFF)

#compiler options
include(cmake/modules/compiler.cmake)
if(CMAKE_BUILD_TYPE STREQUAL "Coverage")
  include(cmake/modules/CodeCoverage.cmake)
  SETUP_TARGET_FOR_COVERAGE(coverage ctest coverage)
endif(CMAKE_BUILD_TYPE STREQUAL "Coverage")

# MFEM
if(MFEM_DIR)
  # MFEM_DIR is modified by FindMFEM.cmake..
  # Don't know why, so in doubt we save the value
  # in your own variable
  set(MFEM_MGIS_MFEM_DIR "${MFEM_DIR}")
endif(MFEM_DIR)
find_package(MFEM REQUIRED)
# MFEM compiled with CUDA: the sources containing kernels of MFEM are
# compiled as CUDA code (see `MFEMMGISDeviceCode.cmake`)
mfem_mgis_enable_device_language()
# CUDA toolkit used to build mfem-mgis, written in its configuration file
if(MFEM_USE_CUDA STREQUAL "YES")
  get_filename_component(MFEM_MGIS_CUDA_TOOLKIT_ROOT "${CUDAToolkit_BIN_DIR}"
                         DIRECTORY)
endif()
# description of MFEM written in the configuration file of mfem-mgis, for
# the projects which find MFEM before mfem-mgis
foreach(_property USE_CUDA USE_HIP CUDA_COMPILER CUDA_HOST_COMPILER
                  CUDA_ARCHITECTURES)
  get_property(MFEM_MGIS_MFEM_${_property} GLOBAL
               PROPERTY MFEM_MGIS_MFEM_${_property})
endforeach(_property)

if (MFEM_USE_MPI)
  message(STATUS "MFEM: using MPI, so mfem_mgis will use MPI")
  find_package(MPI REQUIRED)
  # The following include directories are automatically filled within FindMFEM.cmake
  include_directories(SYSTEM ${HYPRE_INCLUDE_DIRS})
  include_directories(SYSTEM ${METIS_INCLUDE_DIRS})
endif(MFEM_USE_MPI)

# Support for OpenMP
if(MFEM_USE_OPENMP OR MFEM_USE_LEGACY_OPENMP) 
  find_package(OpenMP REQUIRED)
endif(MFEM_USE_OPENMP OR MFEM_USE_LEGACY_OPENMP)

# AmgX, used through its C interface
set(MFEM_MGIS_HAVE_AMGX OFF)
if(enable-amgx)
  if(NOT (MFEM_USE_CUDA STREQUAL "YES"))
    message(FATAL_ERROR "mfem-mgis: AmgX requires MFEM compiled with CUDA")
  endif()
  find_package(AMGX REQUIRED)
  set(MFEM_MGIS_HAVE_AMGX ON)
  get_filename_component(MFEM_MGIS_AMGX_DIR "${AMGX_INCLUDE_DIR}" DIRECTORY)
endif(enable-amgx)

# Kokkos, initialized on demand by mfem_mgis::initialize. The launcher of
# Kokkos only applies to the target compiled by the compiler of Kokkos
# (component separable_compilation, see src/CMakeLists.txt)
set(MFEM_MGIS_HAVE_KOKKOS OFF)
if(enable-kokkos)
  find_package(Kokkos CONFIG REQUIRED COMPONENTS separable_compilation)
  # the behaviours generated by the kokkos backend must share the instance
  # of Kokkos initialized by mfem-mgis
  get_target_property(_kokkos_core_type Kokkos::kokkoscore TYPE)
  if(NOT _kokkos_core_type STREQUAL "SHARED_LIBRARY")
    message(FATAL_ERROR "mfem-mgis: the option enable-kokkos requires "
            "the shared libraries of Kokkos, used by the behaviours "
            "generated by the kokkos backend")
  endif()
  set(MFEM_MGIS_HAVE_KOKKOS ON)
endif(enable-kokkos)

# MFontGenericInterface
find_package (MFrontGenericInterface REQUIRED PATH_SUFFIXES share/mgis/cmake)
# the arrays of the materials are given to MGIS with their location and
# layout, and updated with field operations: the API of MGIS is checked by
# compiling a test (checked again at each configuration, MGIS may change)
include(CheckCXXSourceCompiles)
set(CMAKE_REQUIRED_LIBRARIES mgis::MFrontGenericInterface)
set(CMAKE_REQUIRED_QUIET ON)
unset(MFEM_MGIS_MGIS_HAS_FIELD_OPERATIONS CACHE)
check_cxx_source_compiles([[
#include "MGIS/FieldOperations.hxx"
#include "MGIS/Behaviour/MaterialDataManager.hxx"
int main() {
  auto i = mgis::behaviour::MaterialDataManagerInitializer{};
  i.location = mgis::MemoryLocation::HOST;
  i.layout = mgis::FieldLayout::COMPONENT_BY_COMPONENT;
  auto ops = mgis::HostFieldOperations{};
  static_cast<void>(ops);
  return 0;
}]] MFEM_MGIS_MGIS_HAS_FIELD_OPERATIONS)
unset(CMAKE_REQUIRED_LIBRARIES)
unset(CMAKE_REQUIRED_QUIET)
if(NOT MFEM_MGIS_MGIS_HAS_FIELD_OPERATIONS)
  message(FATAL_ERROR "mfem-mgis requires a version of MGIS providing "
          "field operations (MGIS/FieldOperations.hxx), the location and "
          "the layout of the arrays of the material data managers")
endif()
set(MFEM_MGIS_ENABLE_TFEL_TESTS OFF)
if(MGIS_HAVE_TFEL)
  if(TARGET tfel::TFELTests)
    set(MFEM_MGIS_ENABLE_TFEL_TESTS ON)    
  endif(TARGET tfel::TFELTests) 
endif(MGIS_HAVE_TFEL) 

if(NOT MFEM_MGIS_ENABLE_TFEL_TESTS)
  option(enable-tfel-tests "enable tests based on TFEL tests" OFF)
  if(enable-tfel-tests)
    find_package(TFELTests REQUIRED
      HINTS "${TFEL_DIR}")
    set(MFEM_MGIS_ENABLE_TFEL_TESTS ON)
  endif(enable-tfel-tests)
endif(NOT MFEM_MGIS_ENABLE_TFEL_TESTS)

# find mfront for testing purposes
include(cmake/modules/tfel.cmake)  

# documentation
if (NOT TARGET doc)
  add_custom_target(doc)
endif(NOT TARGET doc)

# add a target to generate API documentation with Doxygen
option(enable-doxygen-doc "enable generation of the Doxygen documentation" OFF)
if(enable-doxygen-doc)
  find_package(Doxygen)
  if(DOXYGEN_FOUND)
    configure_file(${CMAKE_CURRENT_SOURCE_DIR}/Doxyfile.in
      ${CMAKE_CURRENT_BINARY_DIR}/Doxyfile)
    add_custom_target(doc-html
      ${DOXYGEN_EXECUTABLE} ${CMAKE_CURRENT_BINARY_DIR}/Doxyfile
      WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR}
      COMMENT "Generating API documentation with Doxygen" VERBATIM)
    add_dependencies(doc doc-html)
    list(APPEND CPACK_COMPONENTS_ALL doxygen)
    set(CPACK_COMPONENT_DOXYGEN_DESCRIPTION
      "Contains the MFEM/MGIS doxygen documentation")
    set(GENERATE_DOXYGEN ON)
  else(DOXYGEN_FOUND)
    message(FATAL_ERROR "doxygen is required")
  endif(DOXYGEN_FOUND)
else(enable-doxygen-doc)
  message(STATUS "doxygen documentation is not enabled")
endif(enable-doxygen-doc)

add_subdirectory(docs)

# HYPRE version
include(cmake/modules/hypre.cmake)  
set(MFEM_HYPRE_VERSION ${HYPRE_VERSION})
add_compile_definitions(MFEM_HYPRE_VERSION=${MFEM_HYPRE_VERSION})

# testing
set(CTEST_CONFIGURATION_TYPE "${JOB_BUILD_CONFIGURATION}")

add_custom_target(build-tests)
if(CMAKE_CONFIGURATION_TYPES)
   add_custom_target(check COMMAND 
     ${CMAKE_CTEST_COMMAND} -T test --output-on-failure -C $<CONFIGURATION>
     DEPENDS build-tests)
else(CMAKE_CONFIGURATION_TYPES)
   add_custom_target(check COMMAND 
     ${CMAKE_CTEST_COMMAND} -T test --output-on-failure 
     DEPENDS build-tests)
endif(CMAKE_CONFIGURATION_TYPES)

mfem_mgis_buildenv()

add_subdirectory(include)
add_subdirectory(src)
add_subdirectory(tests)
if(BENCHMARKS)
  add_subdirectory(benchmarks)
endif(BENCHMARKS)

# install
install(FILES
   ${PROJECT_SOURCE_DIR}/cmake/modules/FindMFEM.cmake
   ${PROJECT_SOURCE_DIR}/cmake/modules/FindAMGX.cmake
   ${PROJECT_SOURCE_DIR}/cmake/modules/MFEMMGISDeviceCode.cmake
   ${PROJECT_SOURCE_DIR}/cmake/modules/MFEMMGISBehaviours.cmake
   ${PROJECT_SOURCE_DIR}/cmake/modules/ParELAGCMakeUtilities.cmake
   DESTINATION share/mfem-mgis/cmake)
