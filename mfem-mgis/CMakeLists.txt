# Kokkos, initialized on demand by mfem_mgis::initialize. The launcher of
# Kokkos only applies to the target compiled by the compiler of Kokkos
# (component separable_compilation, see src/CMakeLists.txt)
set(MFEM_MGIS_HAVE_KOKKOS OFF)
if(enable-kokkos)
  find_package(Kokkos CONFIG REQUIRED COMPONENTS separable_compilation)
  # the behaviours generated by the kokkos backend must share the instance
  # of Kokkos initialized by mfem-mgis
  get_target_property(_kokkos_core_type Kokkos::kokkoscore TYPE)
  if(NOT _kokkos_core_type STREQUAL "SHARED_LIBRARY")
    message(FATAL_ERROR "mfem-mgis: the option enable-kokkos requires "
            "the shared libraries of Kokkos, used by the behaviours "
            "generated by the kokkos backend")
  endif()
  set(MFEM_MGIS_HAVE_KOKKOS ON)
endif(enable-kokkos)

# Trilinos, used through the shared library MFEMMGISTrilinos, compiled by
# the compiler of Kokkos (see src/CMakeLists.txt)
set(MFEM_MGIS_HAVE_TRILINOS OFF)
if(enable-trilinos)
  if(NOT enable-kokkos)
    message(FATAL_ERROR "mfem-mgis: the option enable-trilinos requires "
            "the option enable-kokkos")
  endif()
  if(NOT MFEM_USE_MPI)
    message(FATAL_ERROR "mfem-mgis: the solvers of Trilinos require MFEM "
            "compiled with MPI (option enable-trilinos)")
  endif()
  find_package(Trilinos CONFIG REQUIRED
               COMPONENTS Tpetra Belos Ifpack2 MueLu Amesos2 ShyLU_DDFROSch)
  # Trilinos must be built with the Kokkos found above: its packages only
  # load their own Kokkos when no target of Kokkos is defined, which would
  # silently compile the sources including Trilinos with another Kokkos
  set(_trilinos_kokkos_config
      "${Trilinos_DIR}/../TeuchosKokkosCompat/TeuchosKokkosCompatConfig.cmake")
  if(EXISTS "${_trilinos_kokkos_config}")
    file(STRINGS "${_trilinos_kokkos_config}" _trilinos_kokkos
         REGEX "set\\(Kokkos_CONFIG ")
    if(_trilinos_kokkos MATCHES "\"(.*)/KokkosConfig.cmake\"")
      get_filename_component(_trilinos_kokkos_dir "${CMAKE_MATCH_1}" REALPATH)
      get_filename_component(_kokkos_dir "${Kokkos_DIR}" REALPATH)
      if(NOT _trilinos_kokkos_dir STREQUAL _kokkos_dir)
        message(FATAL_ERROR "mfem-mgis: Trilinos is built with the Kokkos "
                "of ${_trilinos_kokkos_dir}, mfem-mgis found the Kokkos of "
                "${_kokkos_dir} (option Kokkos_DIR)")
      endif()
    endif()
  endif()
  set(MFEM_MGIS_HAVE_TRILINOS ON)
endif(enable-trilinos)

# sparse part of MAGMA (library magma_sparse and header magmasparse.h),
# searched next to the library of MAGMA which MFEM uses for its batched
# dense algebra (given by its full path, or by a -L flag followed by
# -lmagma), in double precision (the module of mfem-mgis calls the double
# precision interface). The variables MFEM_MGIS_MAGMA_SPARSE_LIBRARY and
# MFEM_MGIS_MAGMA_SPARSE_INCLUDE_DIR may be given to use other locations;
# they are searched again when the library of MAGMA of MFEM changes.
set(MFEM_MGIS_HAVE_MAGMA_SPARSE OFF)
if(MFEM_USE_MAGMA STREQUAL "YES")
  if(MFEM_MGIS_REAL_TYPE STREQUAL "double")
    separate_arguments(_mfem_ext_libs UNIX_COMMAND "${MFEM_EXT_LIBS}")
    set(_mfem_library_directories)
    set(_magma_library_directory)
    foreach(_flag ${_mfem_ext_libs})
      if(_flag MATCHES "^-L(.+)$")
        list(APPEND _mfem_library_directories "${CMAKE_MATCH_1}")
      elseif(_flag MATCHES "^(.+)/libmagma\\.(so|a|dylib)$")
        set(_magma_library_directory "${CMAKE_MATCH_1}")
      endif()
    endforeach()
    if(NOT _magma_library_directory)
      foreach(_directory ${_mfem_library_directories})
        if(EXISTS "${_directory}/libmagma.so" OR
           EXISTS "${_directory}/libmagma.a" OR
           EXISTS "${_directory}/libmagma.dylib")
          set(_magma_library_directory "${_directory}")
          break()
        endif()
      endforeach()
    endif()
    if(NOT "${_magma_library_directory}" STREQUAL
       "${MFEM_MGIS_MAGMA_LIBRARY_DIRECTORY}")
      unset(MFEM_MGIS_MAGMA_SPARSE_LIBRARY CACHE)
      unset(MFEM_MGIS_MAGMA_SPARSE_INCLUDE_DIR CACHE)
      set(MFEM_MGIS_MAGMA_LIBRARY_DIRECTORY "${_magma_library_directory}"
          CACHE INTERNAL "directory of the library of MAGMA used by MFEM")
    endif()
    if(_magma_library_directory)
      find_library(MFEM_MGIS_MAGMA_SPARSE_LIBRARY magma_sparse
                   HINTS "${_magma_library_directory}" NO_DEFAULT_PATH)
      get_filename_component(_magma_directory "${_magma_library_directory}"
                             DIRECTORY)
      find_path(MFEM_MGIS_MAGMA_SPARSE_INCLUDE_DIR magmasparse.h
                HINTS "${_magma_directory}/include" NO_DEFAULT_PATH)
    endif()
    if(MFEM_MGIS_MAGMA_SPARSE_LIBRARY AND MFEM_MGIS_MAGMA_SPARSE_INCLUDE_DIR)
      set(MFEM_MGIS_HAVE_MAGMA_SPARSE ON)
    else()
      message(WARNING "mfem-mgis: MFEM is built with MAGMA, but the "
              "library magma_sparse or the header magmasparse.h is not "
              "found next to the library of MAGMA used by MFEM: the sparse "
              "solvers of MAGMA are not available (see the variables "
              "MFEM_MGIS_MAGMA_SPARSE_LIBRARY and "
              "MFEM_MGIS_MAGMA_SPARSE_INCLUDE_DIR)")
    endif()
  else()
    message(STATUS "mfem-mgis: the sparse solvers of MAGMA are only "
            "available in double precision")
  endif()
endif()
message(STATUS "mfem-mgis sparse solvers of MAGMA: "
        "${MFEM_MGIS_HAVE_MAGMA_SPARSE}")

# HYPRE version
include(cmake/modules/hypre.cmake)  
set(MFEM_HYPRE_VERSION ${HYPRE_VERSION})
add_compile_definitions(MFEM_HYPRE_VERSION=${MFEM_HYPRE_VERSION})
# in single precision, hypre must be built with its internal LAPACK: with an
# external LAPACK, hypre 3.2.0 calls the LAPACK routines of the double type
# on float arrays (ParaSails, Gaussian elimination 98, 99, 198 and 199 of
# BoomerAMG, Schwarz, FSAI, MGR, AIR, GSMG, LOBPCG, AME). Checked again at
# each configuration
if(MFEM_USE_MPI AND (MFEM_MGIS_REAL_TYPE STREQUAL "float"))
  include(CMakePushCheckState)
  cmake_push_check_state(RESET)
  set(CMAKE_REQUIRED_INCLUDES ${HYPRE_INCLUDE_DIRS})
  set(CMAKE_REQUIRED_QUIET ON)
  unset(MFEM_MGIS_HYPRE_LAPACK_MATCHES_REAL_TYPE CACHE)
  check_cxx_source_compiles([[
#include "HYPRE_config.h"
#if defined(HYPRE_SINGLE) && !defined(HYPRE_USING_HYPRE_LAPACK)
#error "hypre in single precision with an external LAPACK"
#endif
int main() { return 0; }]] MFEM_MGIS_HYPRE_LAPACK_MATCHES_REAL_TYPE)
  cmake_pop_check_state()
  if(NOT MFEM_MGIS_HYPRE_LAPACK_MATCHES_REAL_TYPE)
    message(WARNING "mfem-mgis: hypre is built in single precision with an "
            "external LAPACK, whose routines of the double type it calls on "
            "float arrays: ParaSails and some other solvers of hypre give "
            "wrong results. Build hypre with its internal LAPACK (spack: "
            "hypre ~lapack)")
  endif()
endif()
