# short tests: two steps on a coarse mesh; the three linear solvers give the
# same solution
set(norm 3.264041277)
add_test(NAME beam-pcg-amg
  COMMAND mpirun -n 2 ./beam -r 1 -n 2 -ls pcg-amg -en ${norm})
add_test(NAME beam-mumps
  COMMAND mpirun -n 2 ./beam -r 1 -n 2 -ls mumps -en ${norm})
# sequential computations: with UMFPACK, which is not available with MFEM in
# single precision (no SuiteSparse), or with the conjugate gradient of MFEM
# and a Jacobi preconditioner. The comparison of the norm detects a wrong
# solution (the message of the program is checked, whatever its exit
# status).
if(NOT MFEM_USE_SUITESPARSE STREQUAL "NO")
  add_test(NAME beam-umfpack
    COMMAND ./beam -r 1 -n 2 -ls umfpack -no-p -en ${norm})
  add_test(NAME beam-wrong-norm
    COMMAND ./beam -r 1 -n 2 -ls umfpack -no-p -en 3.3)
else()
  add_test(NAME beam-sequential
    COMMAND ./beam -r 1 -n 2 -ls cg-jacobi -lt 1e-12 -no-p -en ${norm})
  add_test(NAME beam-wrong-norm
    COMMAND ./beam -r 1 -n 2 -ls cg-jacobi -lt 1e-12 -no-p -en 3.3)
endif()
set_tests_properties(beam-wrong-norm PROPERTIES
  PASS_REGULAR_EXPRESSION "differs from the expected one")

# the same behaviour generated by the generic-parallel interface on the host
# (openmp and stlpar backends), if TFEL provides the configuration files, and
# on the GPU (cuda backend), integrated on the copies of the arrays on the
# GPU, all the computations being made by kernels
add_generic_parallel_mfront_libraries(beam-behaviours CUDA
  "${MFEM_MGIS_SOURCE_DIR}/tests/IsotropicLinearHardeningPlasticity.mfront")
# the other backends for the GPU which work in mfem-mgis (see
# cases/measure-backends-gpu.sh): cuda by clang, hip by clang and by hipcc on
# the NVIDIA platform, kokkos (Kokkos being initialized by the option
# --kokkos); short tests on the GPU
if(PROTO_GPU_BEHAVIOURS)
  foreach(_gpu_backend cuda-clang:cuda hip-clang-nvidia-platform:hip
                       hip-hipcc:hip kokkos:kokkos)
    string(REPLACE ":" ";" _gb "${_gpu_backend}")
    list(GET _gb 0 _name)
    list(GET _gb 1 _suffix)
    if((_suffix STREQUAL "kokkos") AND (NOT PROTO_WITH_KOKKOS))
      continue()
    endif()
    add_generic_parallel_mfront_library(beam-behaviours ${_name}
      config-${_name}.json ${_suffix}
      "${MFEM_MGIS_SOURCE_DIR}/tests/IsotropicLinearHardeningPlasticity.mfront")
    if(NOT beam-behaviours_${_name}_FOUND)
      continue()
    endif()
    set(_options)
    if(_suffix STREQUAL "kokkos")
      set(_options --kokkos)
    endif()
    add_test(NAME beam-gpu-${_name}
      COMMAND mpirun -n 2 ./beam -r 1 -n 2 -ls pcg-amg -bi Batch
              --device cuda ${_options}
              -l "${CMAKE_CURRENT_BINARY_DIR}/generic-parallel-${_name}/src/libGenericParallelBehaviour-${_suffix}.so"
              -en ${norm})
    set_tests_properties(beam-gpu-${_name} PROPERTIES ENVIRONMENT
      "MFEM_MGIS_GRADIENTS_COMPUTATION=Kernels;MFEM_MGIS_INNER_FORCES_COMPUTATION=Kernels;MFEM_MGIS_TANGENT_OPERATOR_COMPUTATION=Kernels"
      RESOURCE_LOCK gpu)
  endforeach()
endif()
foreach(_backend ${beam-behaviours_BACKENDS})
  set(_library "${CMAKE_CURRENT_BINARY_DIR}/generic-parallel-${_backend}/src/libGenericParallelBehaviour-${_backend}.so")
  if(_backend STREQUAL "cuda")
    add_test(NAME beam-gpu
      COMMAND mpirun -n 2 ./beam -r 1 -n 2 -ls pcg-amg -bi Batch
              --device cuda -l "${_library}" -en ${norm})
    # each process on the GPU reserves a memory pool of 4 GiB (Umpire): the
    # tests on the GPU are not run at the same time
    set_tests_properties(beam-gpu PROPERTIES ENVIRONMENT
      "MFEM_MGIS_GRADIENTS_COMPUTATION=Kernels;MFEM_MGIS_INNER_FORCES_COMPUTATION=Kernels;MFEM_MGIS_TANGENT_OPERATOR_COMPUTATION=Kernels"
      RESOURCE_LOCK gpu)
    # refined beam, ten steps, assembly in a fixed order (same result at
    # each run), compared to the solution in double precision with a tight
    # criterion (2.4e-5 away in single precision, 2.2e-6 in double
    # precision): in single precision, the Newton algorithm diverged at the
    # eighth step when the stiffness kernels were compiled with contractions
    # into fused multiply-add operations, or converged up to 3e-4 away with
    # the atomic assembly (results/diagnostic-float-gpu-2026-10-14)
    add_test(NAME beam-gpu-r4
      COMMAND mpirun -n 1 ./beam -r 4 -n 10 -ls pcg-amg -lt 1e-4 -rt 1e-4
              -bi Batch --device cuda -l "${_library}"
              -en 48.83704586 -nt 5e-5)
    set_tests_properties(beam-gpu-r4 PROPERTIES ENVIRONMENT
      "MFEM_MGIS_GRADIENTS_COMPUTATION=Kernels;MFEM_MGIS_INNER_FORCES_COMPUTATION=Kernels;MFEM_MGIS_TANGENT_OPERATOR_COMPUTATION=Kernels;MFEM_MGIS_TANGENT_OPERATOR_ASSEMBLY=Ordered"
      RESOURCE_LOCK gpu)
  else()
    add_test(NAME beam-batch-${_backend}
      COMMAND mpirun -n 2 ./beam -r 1 -n 2 -ls pcg-amg -bi Batch
              -l "${_library}" -en ${norm})
  endif()
endforeach()
